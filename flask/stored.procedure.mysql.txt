DROP PROCEDURE IF EXISTS ROWPERROW;
DELIMITER ;;

CREATE PROCEDURE ROWPERROW()
BEGIN
DECLARE n INT DEFAULT 0;
DECLARE i INT DEFAULT 0;
SELECT MAX(id) FROM editalProjeto INTO n;
SET i=0;
START TRANSACTION;
WHILE i<=n DO 
    UPDATE editalProjeto SET token=(SELECT substring(MD5(RAND()), -30)) WHERE id=i;
	SET i = i + 1;
END WHILE;
End;
;;

DELIMITER ;
CALL NOMEDAFUNCAO() ;
